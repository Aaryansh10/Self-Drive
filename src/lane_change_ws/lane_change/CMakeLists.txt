cmake_minimum_required(VERSION 3.8)
project(lane_change)
set(CMAKE_CXX_STANDARD 17)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(lane_change_interfaces REQUIRED)
find_package(nav2_msgs QUIET)   # optional: only for the /speed_limit output

add_library(lane_change_core STATIC src/core.cpp)
target_include_directories(lane_change_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)

add_executable(lane_change_server src/lane_change_server.cpp)
target_link_libraries(lane_change_server lane_change_core)
ament_target_dependencies(lane_change_server rclcpp rclcpp_action geometry_msgs nav_msgs std_msgs lane_change_interfaces)
if(nav2_msgs_FOUND)
  ament_target_dependencies(lane_change_server nav2_msgs)
  target_compile_definitions(lane_change_server PRIVATE HAVE_NAV2_MSGS)
  message(STATUS "nav2_msgs found: /speed_limit output enabled")
else()
  message(STATUS "nav2_msgs NOT found: /speed_limit output disabled")
endif()

# offline checks of the planner (no ROS graph needed)
add_executable(path_check tools/path_check.cpp)
target_link_libraries(path_check lane_change_core)
add_executable(sim_check tools/sim_check.cpp)
target_link_libraries(sim_check lane_change_core)

install(TARGETS lane_change_server path_check sim_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
